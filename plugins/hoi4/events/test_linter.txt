# PDX-Mod-Studio Linter 動作検証用テストファイル
# 
# 【検証方法】
# このファイルを PDX-Mod-Studio のテキストエディタタブで開くと、
# リアルタイムで赤い波線（エラー）と黄色の波線（警告）が表示されます。
# 各波線の上にマウスカーソルを乗せると、美しく装飾されたリッチHTMLポップアップが表示されます。

test_event = {
    id = test.1
    
    # 1. 🟡 [警告] 空のコードブロック
    # 中身が空のため、波括弧全体に黄色の警告波線が表示されます。
    trigger = {
    }
    
    immediate = {
        # 🟢 [正常] 正常な記述
        # integer想定に対して正しい整数値が入っているため、波線は出ません。
        add_threat = 10
        
        # 2. 🟡 [警告] 整数想定の場所に小数を入力
        # integer想定のadd_manpowerにドット(.)付き小数を書いたため、黄色い警告波線が表示されます。
        add_manpower = 100.5
        
        # 3. 🔴 [エラー] 致命的な型ミスマッチ
        # 整数が必要なadd_manpowerに、全く異なる真偽値(yes)を指定したため、赤いエラー波線が表示されます。
        add_manpower = yes
        
        # 4. 🔴 [エラー] 未定義のエフェクト（スペルミスなど）
        # effects.jsonに存在しないキー名のため、赤いエラー波線が表示されます。
        add_thr = 5
        
        # 5. 🟡 [警告] 非推奨（Deprecated）のエフェクト使用
        # effects.jsonで非推奨に指定されたエフェクトのため、黄色い警告波線が表示されます。
        # ホバーすると、「推奨される代替案: 代わりに add_threat を使用してください」と表示されます。
        add_threat_old = 15
        
        # 6. 🟡 [警告] 物理限界値 (±2,147,483) の超過
        # ゲーム共通の限界値を超えた数値を入力したため、黄色いオーバーフロー警告波線が表示されます。
        add_manpower = 3000000
        
        # 7. 🟡 [警告] 小数有効精度 (5桁) の切り捨て
        # 小数点第5位を超える細かすぎる小数を書いたため、黄色い警告波線が表示されます。
        add_threat = 5.123456
    }
}
